# Root: location. Room left under the 100% control ceiling.
glorpui_control_headroom = {
	value = 1
	subtract = max_control
	min = 0
}

# Root: location. Expects scopes: glorpui_control_building_type.
# Control one level of the building type adds here.
glorpui_control_building_gain = {
	value = 0
	add = {
		scope:glorpui_control_building_type = {
			value = glorpui_control_building_max_control
		}
	}
	max = glorpui_control_headroom
}

# Root: location. Expects scopes: glorpui_control_building_type, glorpui_loc_tax_income.
# Monthly tax income the added control brings in, at the rate the location is taxed at today.
glorpui_control_building_income = {
	value = 0
	if = { limit = { local_control > 0 }
		add = {
			value = scope:glorpui_loc_tax_income
			multiply = glorpui_control_building_gain
			divide = local_control
		}
	}
}
