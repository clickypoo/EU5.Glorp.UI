SetUseVanillaPortrait = {
    effect = {
        if = {
            limit = { NOT = { has_variable = useVanillaPortrait } }
            set_variable = { name = useVanillaPortrait value = 1 }
        }
        else = {
            #set_variable = { name = useVanillaPortrait value = 1 }
            remove_variable = useVanillaPortrait
        }
    }
}
